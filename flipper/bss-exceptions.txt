# Base programs allowed above 4 KB of .bss (spec R2a.5), checked by
# tools/bsscheck.py in "make ramcheck".  Format: name max-bytes  # reason
# Lower a line when the program shrinks; add one only with the reason.
#
# lat: diagnostic, only useful on the latency build (make latcheck).  It
# keeps 1024 kernel events (8 KB) from a whole "lat -w SECONDS" run so
# that p99/max are computed over all of them; a smaller ring would drop
# events from the L1-L10 measurements.
lat	9832
# ir: one raw IR frame, 1024 durations of 32 bits (4 KB).  The same
# buffer is the /dev/ir transmit format (first word: carrier), and long
# air-conditioner frames need several hundred durations.
ir	4344
# launcher: the desktop (spec K.6, K.8, T23b).  Its window server keeps
# the window on the screen and its parent (2 x 658 B), a record line
# (128 B) and the keyboard's text (64 B); the clock window's localtime()
# brings libc's TZ state (~400 B).  The rest is the frame buffer (1 KB)
# and getpwnam/getgrnam's line buffers.  K.6 budget: <= 3 KB above the
# launcher before the desktop (.bss +1.9 KB, .data +0.4 KB here).
# T17a (#249): +4 B, fsck_err (the "card has errors" title).
# T23a (K.6): +0.8 KB, the desktop's tile table (16 x 48 B) and its state;
# -1.3 KB, getgrnam()'s buffers (the gid of "gui" is read on the stack).
# Data + bss: 6196 B, +1864 B above the launcher before the desktop
# (4332 B at d704a5c), inside the K.6 budget of 3 KB.
# T23c (K.6): +680 B, the last 8 notifications (8 x 88 B, in place of the
# one 32 B text) and the opened one's index; .data +24 B (GOT slots).  A
# malloc'd table does not fit: the launcher's 4 KB gap has no 704 B free
# (nattest).  Data + bss: 6928 B (main 1cc7a8f: 6224 B), +2596 B above
# the launcher before the desktop, inside the K.6 budget of 3 KB.
launcher	5468
# mtpd: files to a PC over MTP (spec R10.4, T29a).  The spec's parent ring
# (256 x 8 B = 2 KB) and one-block buffer (1 KB), two 256 B paths, and its
# one DIR (1 KB) kept here instead of opendir()'s malloc: no heap, so no
# MM gap.  Running: data 684 + .bss + CHMEM 2048 = 7.4 KB (spec <= 8 KB).
mtpd	4828
